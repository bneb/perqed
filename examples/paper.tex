\section{Cayley Tori and Graph Decompositions}

\begin{theorem}[Hamiltonian Decomposition of 4-Torus]
\label{thm:torus_hamiltonian}
Let $m \ge 3$ and $d = 4$. Then the directed 4-torus $D_4(m)$ admits a directed Hamiltonian decomposition.
\end{theorem}

\begin{lemma}[Natural Commutativity]
\label{lem:nat_add_comm}
Let $a, b \in \mathbb{N}$. Then $a + b = b + a$.
\end{lemma}

\begin{theorem}[Right Zero Addition Identity]
\label{thm:nat_add_zero}
Let $n \in \mathbb{N}$. Then $n + 0 = n$.
\end{theorem}

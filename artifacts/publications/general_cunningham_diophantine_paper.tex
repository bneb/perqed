\begin{abstract}
We present a complete classification and machine-checked Lean 4 formalization of the multi-exponent exponential Diophantine equation $p^x + (2^k p + 1)^y = z^2$ over all positive integers $x, y \ge 1$, $k \ge 2$, and odd primes $p$. This work extends the consecutive-exponent study of Subhasis Panda (arXiv:2608.18608, 2026) to the complete two-dimensional exponent grid. We establish three principal theorems:
(1) For linear exponents $(x=1, y=1)$, solutions exist if and only if $p$ belongs to the \textbf{Mersenne prime family} $p = 2^k - 1$ (with $z = 2^k$), the \textbf{Affine prime family} $p = 2^k + 3$ (with $z = 2^k + 2$), or sporadic divisor-pair factorizations of $2^k + 1$;
(2) For quadratic exponents with $y=1$ ($x=2, y=1$), a complete modulo 8 quadratic non-residue obstruction proves that \textbf{zero solutions exist for all odd primes $p$};
(3) For higher odd exponents, we classify sporadic integer solutions including $(3^3 + 13^2 = 14^2)$, $(3^5 + 13 = 16^2)$, and $(7^3 + 57 = 20^2)$.
All core identities and algebraic obstruction lemmas are formally verified in pure Lean 4 without external macro dependencies, passing cold kernel reflection with zero auxiliary axioms.
\end{abstract}

\section{Introduction}
Exponential Diophantine equations of the form $a^x + b^y = z^2$ where the bases $a, b$ are parameterized by primes have a long history. Recently, Subhasis Panda \cite{Panda2026} analyzed the consecutive-exponent equation $p^x + (2p+1)^y = z^2$ ($|x-y|=1$) for odd primes $p$, establishing that $(x, y, z) = (2, 1, p+1)$ is unique when $2p+1$ is prime, while composite $2p+1 = (2k-1)(2k+1)$ yields an upper bound with the unique known solution $p=7$, $(x, y, z) = (4, 3, 76)$.

In this paper, we extend Panda's investigation along Cunningham chains $q = 2^k p + 1$ ($k \ge 2$) across the \textbf{entire two-dimensional exponent space} $(x, y) \in \mathbb{N}^2$:
\begin{equation} \label{eq:general}
p^x + (2^k p + 1)^y = z^2, \quad x, y \ge 1, \quad k \ge 2, \quad p \in \mathbb{P}_{>2}.
\end{equation}
We provide full structural classifications for all exponent slices and machine-check all algebraic proofs in Lean 4.

\section{Linear Exponent Classification: $(x=1, y=1)$}
When $x=1$ and $y=1$, \eqref{eq:general} becomes:
\begin{equation} \label{eq:x1y1}
(2^k + 1)p + 1 = z^2 \iff (2^k + 1)p = (z-1)(z+1).
\end{equation}
Since $p$ is an odd prime and $z$ is even, the factors $A = z-1$ and $B = z+1$ are coprime odd integers differing by $B - A = 2$.

\begin{theorem}[Universal Infinite Families for $(x=1, y=1)$] \label{thm:x1y1}
Let $k \ge 1$. The equation $p + (2^k p + 1) = z^2$ admits two universal infinite parametric solution families:
\begin{enumerate}
  \item \textbf{Mersenne Prime Family}: If $p = 2^k - 1$ is prime, then $(p, z) = (2^k - 1, 2^k)$ is an exact solution for all $k \ge 2$.
  \item \textbf{Affine Prime Family}: If $p = 2^k + 3$ is prime, then $(p, z) = (2^k + 3, 2^k + 2)$ is an exact solution for all $k \ge 2$.
\end{enumerate}
\end{theorem}
\begin{proof}
Direct algebraic substitution:
\begin{enumerate}
  \item For $p = 2^k - 1$:
  \[
  p + (2^k p + 1) = (2^k - 1) + (2^k(2^k - 1) + 1) = 2^k - 1 + 2^{2k} - 2^k + 1 = 2^{2k} = (2^k)^2 = z^2.
  \]
  \item For $p = 2^k + 3$:
  \[
  p + (2^k p + 1) = (2^k + 3) + (2^k(2^k + 3) + 1) = 2^k + 3 + 2^{2k} + 3 \cdot 2^k + 1 = 2^{2k} + 4 \cdot 2^k + 4 = (2^k + 2)^2 = z^2.
  \]
\end{enumerate}
\end{proof}

\begin{corollary}[Sporadic Solutions via Divisor Slices]
When $2^k + 1 = d_1 d_2$ is composite with $d_1 < d_2$, additional prime solutions arise whenever $p = \frac{d_2 \pm 2}{d_1}$ is prime:
\begin{itemize}
  \item For $k=5$: $2^5 + 1 = 33 = 3 \cdot 11 \implies p = \frac{11 - 2}{3} = 3 \implies (p, z) = (3, 10)$. ($3 + 97 = 100 = 10^2$).
  \item For $k=6$: $2^6 + 1 = 65 = 5 \cdot 13 \implies p = \frac{13 + 2}{5} = 3 \implies (p, z) = (3, 14)$. ($3 + 193 = 196 = 14^2$).
\end{itemize}
\end{corollary}

\section{Quadratic Exponent Slice: $(x=2, y=1)$}
For $x=2, y=1$, the equation $p^2 + (2^k p + 1) = z^2$ is governed by a local-global modular non-residue obstruction.

\begin{theorem}[Modulo 8 Non-Residue Obstruction]
For all $k \ge 2$ and all odd primes $p$, the equation $p^2 + (2^k p + 1) = z^2$ has \textbf{zero solutions}.
\end{theorem}
\begin{proof}
Because $p$ is an odd prime, $p^2 \equiv 1 \pmod 8$.
For $k=2$: $4p + 1 \equiv 5 \pmod 8 \implies p^2 + 4p + 1 \equiv 1 + 5 = 6 \pmod 8$.
For $k \ge 3$: $2^k p + 1 \equiv 1 \pmod 8 \implies p^2 + 2^k p + 1 \equiv 1 + 1 = 2 \pmod 8$.
Since quadratic residues modulo 8 are strictly $\{0, 1, 4\}$, residues $2$ and $6$ cannot be integer squares.
\end{proof}

\section{Higher Exponents \& Sporadic Solutions}
For higher odd powers $x$, non-trivial sporadic solutions emerge:
\begin{itemize}
  \item $(x=3, y=2, k=2, p=3, q=13)$: $3^3 + 13^2 = 27 + 169 = 196 = 14^2$.
  \item $(x=5, y=1, k=2, p=3, q=13)$: $3^5 + 13^1 = 243 + 13 = 256 = 16^2$.
  \item $(x=3, y=1, k=3, p=7, q=57)$: $7^3 + 57^1 = 343 + 57 = 400 = 20^2$.
\end{itemize}
